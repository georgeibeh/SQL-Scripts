USE UKTBDP;

SELECT

[APR_CXR_STATUS] CXR_STATUS

,CASE

 WHEN [APR_XRAY_GIVEN_TO_APP] ='Y' THEN 'Yes'

 WHEN [APR_XRAY_GIVEN_TO_APP] ='N' THEN 'No'

 END XRAY_GIVEN_TO_APP

,[APR_LOCAL_CURRENCY] LOCAL_CURRENCY

,CASE

WHEN [APR_PAY_FROM_UK_REFUGEE] ='Y' THEN 'Yes'

WHEN [APR_PAY_FROM_UK_REFUGEE] ='N' THEN 'No'

END PAY_FROM_UK_REFUGEE

,[APR_PAY_LOCALAMOUNT] PAY_LOCALAMOUNT

,CAST([APR_PAY_DATE_PAID] as DATE) PAY_DATE_PAID

,[APR_PAY_DETAILS] PAY_DETAILS

,[APR_PM_NAME] PM_NAME

,[APR_PAY_RECEIPT_NUMBER] PAY_RECEIPT_NUMBER

,CAST([APR_DATE_FIRST_CONTACT] as DATE) DATE_FIRST_CONTACT

,CASE

WHEN [APR_HAS_FAMILY_WITH_TB] = 'Y' THEN 'Yes'

WHEN [APR_HAS_FAMILY_WITH_TB] = 'N' THEN 'No'

END HAS_FAMILY_WITH_TB

,[APR_NATIONALITY_CNT_NAME] NATIONALITY_CNT_NAME

,CAST([APR_PASSPORT_DATE_EXPIRES] as DATE) PASSPORT_DATE_EXPIRES

,CAST([APR_PASSPORT_DATE_ISSUED] as DATE) PASSPORT_DATE_ISSUED

,[APR_PASSPORT_NUMBER] PASSPORT_NUMBER

,[APR_REFERENCE_ID] REFERENCE_ID

,[APR_LOC_ADDRESS_LINE2] LOC_ADDRESS_LINE2

,[APR_LOC_CTY_NAME] LOC_CTY_NAME

,[APR_LOC_NAME] LOC_NAME

,[APR_ADDRESS1] ADDRESS1

,[APR_ADDRESS2] ADDRESS2

,CAST([APR_BIRTHDATE] as DATE) BIRTHDATE

,(DATEDIFF(year,[APR_BIRTHDATE],[APR_APPOINTMENT_DATE])) Age

,[APR_EMAIL] EMAIL

,[APR_FIRSTNAME] FIRSTNAME

,CASE

WHEN [APR_GENDER]= 'M' THEN 'Male'

ELSE 'Female'

END GENDER

,[APR_LASTNAME] LASTNAME

,[APR_MIDDLENAME] MIDDLENAME

,[APR_TELEPHONE] TELEPHONE

,CASE

WHEN [APR_SO_CERT_ISSUED] ='Y' THEN 'Yes'

WHEN [APR_SO_CERT_ISSUED] ='N'THEN 'No'

END SO_CERT_ISSUED

,[APR_SO_NOT_REASON] SO_NOT_REASON

,[APR_SO_NOT_OTHERS] SO_NOT_OTHERS

,[APR_VT_NAME] VISA_TYPE

,[APR_MC_ISSUEDBY] MC_ISSUEDBY

,CAST([APR_EXAM_DATE] as DATE) EXAM_DATE

,DATENAME(month,[APR_EXAM_DATE]) EXAM_Month

,CASE

WHEN [APR_MC_REISSUE] ='Y' THEN 'Yes'

WHEN [APR_MC_REISSUE] ='N' THEN 'No'

END REISSUE

,CASE

WHEN [APR_SS_IS_CERTIFIED]  ='Y' THEN 'Yes'

WHEN [APR_SS_IS_CERTIFIED]  ='N' THEN 'No'

END SSIS_CERTIFIED

,CAST([APR_SS_DATE_CERTIFIED] as DATE) SS_DATE_CERTIFIED

,[APR_SS_FINAL_STATUS] SS_FINAL_STATUS

,[APR_SS_UNDONE_OTHER] SS_UNDONE_OTHER

,[APR_SS_CERTIFIEDBY] SS_CERTIFIEDBY

,CASE

WHEN [APR_SS_IS_CONFIRMED] ='Y' THEN 'Yes'

WHEN [APR_SS_IS_CONFIRMED] ='N' THEN 'No'

END SS_IS_CONFIRMED

,CASE

WHEN [APR_IS_REFERRAL_ISSUED] ='Y' THEN 'Yes'

WHEN [APR_IS_REFERRAL_ISSUED] ='N' THEN 'No'

END IS_REFERRAL_ISSUED

,[APR_PAY_CERTIFIEDBY] PAY_CERTIFIEDBY

,CASE

WHEN [APR_HAS_TB_HISTORY]= 'Y' THEN 'Yes'

WHEN [APR_HAS_TB_HISTORY]= 'N' THEN 'No'

END HAS_TB_HISTORY

,CASE

WHEN [APR_PREV_TB_TREATED] = 'Y' THEN 'Yes'

WHEN [APR_PREV_TB_TREATED] = 'N' THEN 'No'

END PREV_TB_TREATED

,[APR_MC_HOLOGRAM] HOLOGRAM_NUMBER

,CAST([APR_MC_ORIG_DATE_ISSUED] as DATE) MC_ORIG_DATE_ISSUED

,CAST([APR_APPOINTMENT_DATE] as DATE) APPOINTMENT_DATE

,DATENAME(month,[APR_APPOINTMENT_DATE]) Appointment_Month

,CAST([APR_DATE_VERIFIED] as DATE) DATE_VERIFIED

,CAST([APR_SO_DATE_VERIFIED] as DATE) SO_DATE_VERIFIED

,[APR_SO_CERTIFIED_BY] SO_CERTIFIED_BY

,CAST([APR_CXR_DATE_DONE] as DATE) CXR_DATE_DONE

,[APR_MC_ISSUEDBY_FOR_EXPORT] MC_ISSUEDBY_FOR_EXPORT

,[APR_REVOKED] REVOKED

,CAST([APR_DATE_VERIFIED_FIRST] as DATE) DATE_VERIFIED_FIRST

,CAST([APR_SO_DATE_VERIFIED_FIRST] as DATE) SO_DATE_VERIFIED_FIRST

,CASE CONVERT(varchar, DATEPART(HOUR, [APR_DATE_VERIFIED_FIRST]))
    WHEN '6' THEN   '7:00 AM'
    WHEN '7' THEN   '8:00 AM'
    WHEN '8' THEN   '9:00 AM'
    WHEN '9' THEN   '10:00 AM'
    WHEN '10' THEN  '11:00 AM'
    WHEN '11' THEN  '12:00 PM'
    WHEN '12' THEN  '1:00 PM'
    WHEN '13' THEN  '2:00 PM'
    WHEN '14' THEN  '3:00 PM'
    WHEN '15' THEN  '4:00 PM'
    WHEN '16' THEN   '5:00 PM'
    WHEN '17' THEN  '6:00 PM'
    WHEN '18' THEN '7:00 PM'
    ELSE  CONVERT(varchar,DATEPART(HOUR,[APR_DATE_VERIFIED_FIRST]))
END Hour_Verified

,[APR_VERIFIED_BY] VERIFIED_BY

,[APR_CONTACT_MODE] CONTACT_MODE

,[APR_UKCONTACT_ADDRESS] UKCONTACT_ADDRESS

,[APR_UKCONTACT_POSTCODE] UKCONTACT_POSTCODE

,[APR_UKCONTACT_TELEPHONE] UKCONTACT_TELEPHONE

,CASE

WHEN [APR_UKSPONSOR_IS_UK_SPONSORED] ='Y' THEN 'Yes'

WHEN [APR_UKSPONSOR_IS_UK_SPONSORED] ='N' THEN 'No'

END IS_UK_SPONSORED

,[APR_UKSPONSOR_NAME] UKSPONSOR_NAME

,[APR_UKSPONSOR_ADDRESS] UKSPONSOR_ADDRESS

,[APR_UKSPONSOR_POSTCODE] UKSPONSOR_POSTCODE

,[APR_UKSPONSOR_TELEPHONE] UKSPONSOR_TELEPHONE

,CASE

WHEN [APR_UKSPONSOR_IS_TYPE_EDUC] ='Y' THEN 'Yes'

WHEN [APR_UKSPONSOR_IS_TYPE_EDUC] ='N' THEN 'No'

END UKSPONSOR_IS_TYPE_EDUC

,CASE

WHEN [APR_UKSPONSOR_IS_TYPE_EMP] ='Y' THEN 'Yes'

WHEN [APR_UKSPONSOR_IS_TYPE_EMP] ='N' THEN 'No'

END UKSPONSOR_IS_TYPE_EMP

,CASE

WHEN [APR_UKSPONSOR_IS_TYPE_FAM] ='Y' THEN 'Yes'

WHEN [APR_UKSPONSOR_IS_TYPE_FAM] ='N' THEN 'No'

 END UKSPONSOR_IS_TYPE_FAM

,CASE

WHEN [APR_UKSPONSOR_IS_TYPE_OTH] ='Y' THEN 'Yes'

WHEN [APR_UKSPONSOR_IS_TYPE_OTH] ='N' THEN 'No'

END UKSPONSOR_IS_TYPE_OTH

,[APR_UKSPONSOR_TYPE_DETAILS] UKSPONSOR_TYPE_DETAILS

,[APR_IS_PE_DONE] IS_PE_DONE

,[APR_MOBILEPHONE] MOBILEPHONE

  FROM [UKTBDP].[dbo].[APPL_REPORTS]

  WHERE [APR_LOC_CNT_CODE] ='NG' AND [APR_EXAM_DATE] = CAST(GETDATE() AS DATE);
