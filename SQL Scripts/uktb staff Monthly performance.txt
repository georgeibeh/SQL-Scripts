USE UKTBDP;

SELECT
[APR_PAY_CERTIFIEDBY]
,[JANUARY], [FEBRUARY], [MARCH], [APRIL], [MAY], [JUNE], [JULY], [AUGUST]
FROM
(  SELECT
   [APR_PAY_CERTIFIEDBY]
   ,DATENAME(month,[APR_EXAM_DATE]) [Month]
   ,[APR_VERIFIED_BY]
    FROM [UKTBDP].[dbo].[APPL_REPORTS]
    WHERE [APR_LOC_CNT_CODE] ='NG' AND [APR_EXAM_DATE] LIKE '2023%' AND [APR_PAY_CERTIFIEDBY] IS NOT NULL
) AS SRC
  PIVOT(
         COUNT([APR_VERIFIED_BY]) 
         FOR [Month] IN ([JANUARY], [FEBRUARY], [MARCH], [APRIL], [MAY], [JUNE], [JULY], [AUGUST])
         ) AS PVT;
